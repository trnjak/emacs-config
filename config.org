#+TITLE: My Emacs Config
#+PROPERTY: header-args:emacs-lisp :lexical t

* About
This was made as an attempt to make Emacs more usable on modern macOS, as well as make it look cool.\\
While it consists of things that are useful to me, the configuration is very minimal and modular, so anyone can easily build upon it. :)

Inspired by [[https://github.com/snackon/Witchmacs][Witchmacs]]. It's what got me into Emacs back when I had a laptop that ran GNU/Linux. While Witchmacs is pretty outdated now, I recommend checking it out, as it explains things in greater detail.

* Install
Once Emacs is installed on your system, simply run the code below in your terminal:

#+BEGIN_SRC shell
  git clone https://github.com/trnjak/emacs-config.git ~/.emacs.d/
#+END_SRC

After installing, the first run of Emacs will take a bit longer, as it has to install the packages listed in the README.org file for the first time.

* Configuration
** Optimisations for macOS
#+BEGIN_SRC emacs-lisp
;;; -*- lexical-binding: t; -*-

(setq mac-option-modifier 'meta)
(setq mac-command-modifier 'super)

(use-package exec-path-from-shell
  :ensure t
  :init
  (setq exec-path-from-shell-arguments nil)
  :config
  (when (or (memq window-system '(mac ns x))
            (eq system-type 'darwin)
            (daemonp))
    (exec-path-from-shell-initialize)))
#+END_SRC

  (interactive)
  (split-window-below)
  (balance-windows)
  (other-window 1))
(global-set-key (kbd "C-x 2") #'split-and-follow-horizontally)

(defun split-and-follow-vertically ()
  (interactive)
  (split-window-right)

** dashboard
#+BEGIN_SRC emacs-lisp
(use-package dashboard
  :ensure t
  :init
  (setq initial-buffer-choice (lambda () (get-buffer-create "*dashboard*")))
  :config
  (dashboard-setup-startup-hook)
  (setq dashboard-banner-logo-title nil)
  (let ((banner-img (expand-file-name "dash.png" user-emacs-directory)))
    (setq dashboard-startup-banner
          (if (and (display-graphic-p) (file-exists-p banner-img))
              banner-img
            '1)))
  (setq dashboard-items '((recents . 5)))
  (setq dashboard-center-content t)
  (setq dashboard-items-default-alignment 'center)
  (setq dashboard-icon-type 'nerd-icons)
  (setq dashboard-set-heading-icons t)
  (setq dashboard-set-file-icons t))
#+END_SRC

** eglot
#+BEGIN_SRC emacs-lisp
(when-let* ((npm-root (ignore-errors
                        (string-trim
                         (shell-command-to-string "npm root -g 2>/dev/null")))))
  (when (file-directory-p npm-root)
    (setenv "NODE_PATH"
            (if (getenv "NODE_PATH")
                (concat npm-root ":" (getenv "NODE_PATH"))
              npm-root))))

(use-package eglot
  :ensure t
  :defer t
  :hook
  ((c-mode . eglot-ensure)
   (c-ts-mode . eglot-ensure)
   (c++-mode . eglot-ensure)
   (c++-ts-mode . eglot-ensure)
   (java-mode . eglot-ensure)
   (java-ts-mode . eglot-ensure)
   (rust-mode . eglot-ensure)
   (rust-ts-mode . eglot-ensure)
   (web-mode . eglot-ensure)
   (html-mode . eglot-ensure)
   (css-mode . eglot-ensure)
   (css-ts-mode . eglot-ensure)
   (js-mode . eglot-ensure)
   (js-ts-mode . eglot-ensure)
   (typescript-mode . eglot-ensure)
   (typescript-ts-mode . eglot-ensure)
   (python-mode . eglot-ensure)
